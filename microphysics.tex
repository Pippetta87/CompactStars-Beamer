\section{3.1 Potenziali termodinamici}

\begin{frame}[t]{Ambito e convenzioni}
\small

Il capitolo 3 descrive materia densa in equilibrio: potenziali, composizione in $\beta$-equilibrio, gas fermionici e politropiche. L'estensione finale tratta la \emph{superfluidit\`a neutronica}, non presente nel capitolo allegato.
\begin{itemize}
\item Unita naturali $\hbar=c=k_B=1$ salvo indicazione contraria; $\varepsilon=U/V$.
\item $N_i$ numero di particelle della specie $i$; $n_i=N_i/V$, $s=S/V$.
\item $\mu_i$ include l'energia a riposo e si riferisce al numero conservato appropriato.
\item Gli integrali dei gas liberi presuppongono particelle omogenee a $T=0$.
\end{itemize}
Distinguere sistematicamente: $P(n)$, $P(\varepsilon)$ e $P(\mu)$ non contengono la stessa informazione termodinamica.
\end{frame}

\begin{frame}[t]{Primo principio e variabili naturali}
\small

\[
dU=T\,dS-P\,dV+\sum_i\mu_i\,dN_i,
\qquad U=U(S,V,\{N_i\}).
\]
\[
T=\left.\frac{\partial U}{\partial S}\right|_{V,N_i},\quad
P=-\left.\frac{\partial U}{\partial V}\right|_{S,N_i},\quad
\mu_i=\left.\frac{\partial U}{\partial N_i}\right|_{S,V,N_{j\ne i}}.
\]
\begin{itemize}
\item Microcanonico: dati $U,V,N_i$; il principio di equilibrio \`e il massimo dell'entropia $S$ a vincoli fissati.
\item Canonico: dati $T,V,N_i$; si usa l'energia libera di Helmholtz $F$.
\item Gran canonico: dati $T,V,\mu_i$; si usa il grande potenziale $\Omega$.
\end{itemize}
La scelta di un potenziale \`e una trasformazione di Legendre, non un cambio di leggi fisiche.
\end{frame}

\begin{frame}[t]{Derivazione dell'equazione di Eulero}
\small

L'estensivit\`a impone, per un sistema omogeneo bulk,
\[
U(\lambda S,\lambda V,\{\lambda N_i\})=\lambda U(S,V,\{N_i\}).
\]
Deriviamo rispetto a $\lambda$ e poniamo $\lambda=1$:
\[
U=S\frac{\partial U}{\partial S}+V\frac{\partial U}{\partial V}
 +\sum_iN_i\frac{\partial U}{\partial N_i}.
\]
Usando il primo principio,
\[
\boxed{U=TS-PV+\sum_i\mu_iN_i.}
\]
Validita: equilibrio estensivo; superfici, gravitazione auto-consistente e campi a lungo raggio possono richiedere cautela.
\end{frame}

\begin{frame}[t]{Ensemble canonico: trasformazione di Legendre}
\small

\[
F(T,V,N_i)=U-TS.
\]
Differenziale esplicito:
\[
\begin{aligned}
dF&=dU-T\,dS-S\,dT\\
&=-P\,dV-S\,dT+\sum_i\mu_i\,dN_i.
\end{aligned}
\]
\[
P=-\left.\frac{\partial F}{\partial V}\right|_{T,N_i},\quad
S=-\left.\frac{\partial F}{\partial T}\right|_{V,N_i},\quad
\mu_i=\left.\frac{\partial F}{\partial N_i}\right|_{T,V,N_{j\ne i}}.
\]
Per uno stato uniforme $F/V=f(T,n_i)$ e $P=-f+\sum_i\mu_i n_i$.

\end{frame}

\begin{frame}[t]{Ensemble gran canonico: pressione come potenziale}
\small

\[
\Omega(T,V,\mu_i)=F-\sum_i\mu_iN_i=U-TS-\sum_i\mu_iN_i.
\]
\[
\begin{aligned}
d\Omega&=-P\,dV-S\,dT-\sum_iN_i\,d\mu_i,\\
\Omega&=-PV\qquad\text{(bulk, via Eulero).}
\end{aligned}
\]
Dunque
\[
P(T,\mu_i)=-\frac{\Omega}{V},\qquad
s=\left.\frac{\partial P}{\partial T}\right|_{\mu_i},\qquad
n_i=\left.\frac{\partial P}{\partial\mu_i}\right|_{T,\mu_{j\ne i}}.
\]
In particolare $P(\mu,T)$, se termodinamicamente consistente, permette di ricostruire anche densita ed energia.
\end{frame}

\begin{frame}[t]{Limite termodinamico: identita fondamentali}
\small

Per materia bulk omogenea:
\[
\varepsilon=\frac UV,\qquad s=\frac SV,\qquad n_i=\frac{N_i}{V}.
\]
Il primo principio diventa
\[
d\varepsilon=T\,ds+\sum_i\mu_i\,dn_i,
\]
mentre l'equazione di Eulero diventa
\[
\boxed{\varepsilon+P=Ts+\sum_i\mu_i n_i.}
\]
Con $T=0$ e un solo numero conservato indipendente:
\[
d\varepsilon=\mu\,dn,\qquad
\varepsilon+P=\mu n,\qquad dP=n\,d\mu.
\]
Quest'ultima deriva differenziando l'identita di Eulero e confrontando con il primo principio.
\end{frame}

\begin{frame}[t]{Energia per particella e pressione}
\small

Con $T=0$, un numero conservato $N$ e $e(n)=\varepsilon(n)/n$:
\[
\frac{de}{dn}=
\frac{n\,d\varepsilon/dn-\varepsilon}{n^2}.
\]
Poiche $d\varepsilon/dn=\mu$ e $P=\mu n-\varepsilon$,
\[
\boxed{P=n^2\frac{d}{dn}\!\left(\frac{\varepsilon}{n}\right).}
\]
Ne segue
\[
\mu=e+\frac{P}{n},\qquad
P=0\ \Longleftrightarrow\ \frac{de}{dn}=0\quad(n>0).
\]
Un minimo di $e(n)$ a densita finita e a pressione nulla individua materia autolegata; la stabilita richiede inoltre opportuna curvatura positiva.
\end{frame}

\begin{frame}[t]{Ricostruire l'EOS da P(n)}
\small

Da $P=n^2d(\varepsilon/n)/dn$:
\[
\frac{d}{dn}\left(\frac{\varepsilon}{n}\right)=\frac{P(n)}{n^2}.
\]
Integrando fra due densita:
\[
\boxed{\varepsilon(n)=n\left[\frac{\varepsilon(n_*)}{n_*}
+\int_{n_*}^{n}\frac{P(n')}{n'^2}\,dn'\right].}
\]
Per $P(n)$ serve quindi una condizione energetica al punto di riferimento $n_*$.
Una sola relazione $P(\varepsilon)$, sufficiente a integrare le TOV, non determina automaticamente $n(\varepsilon)$ senza un'ulteriore normalizzazione termodinamica.
\end{frame}

\begin{frame}[t]{Caso canonico nel bulk: verifica}
\small

\[
f(T,n_i)=\frac FV=\varepsilon-Ts.
\]
\[
df=-s\,dT+\sum_i\mu_i\,dn_i,
\]
\[
s=-\frac{\partial f}{\partial T},\qquad
\mu_i=\frac{\partial f}{\partial n_i},\qquad
P=-f+\sum_i\mu_i n_i.
\]
Per una sola specie a $T=0$, $f=\varepsilon$ e
\[
P=n\frac{d\varepsilon}{dn}-\varepsilon.
\]
La seconda derivata di $\varepsilon(n)$ controlla la comprimibilita: $dP/dn=n\,d\mu/dn$.
\end{frame}

\section{3.2 Equilibrio chimico}

\begin{frame}[t]{Reazione chimica: condizione generale}
\small

Scriviamo una reazione con coefficienti stechiometrici $\nu_i$ (positivi per i prodotti):
\[
\sum_i\nu_i A_i=0,\qquad dN_i=\nu_i\,d\xi,
\]
dove $\xi$ e l'avanzamento della reazione.
A $T,V$ fissati,
\[
dF=\sum_i\mu_i dN_i=\left(\sum_i\nu_i\mu_i\right)d\xi.
\]
All'equilibrio chimico, consentendo variazioni $d\xi$ di entrambi i segni,
\[
\boxed{\sum_i\nu_i\mu_i=0.}
\]
Questa derivazione evita l'uso ambiguo di $dU=0$ senza esplicitare i vincoli termodinamici.
\end{frame}

\begin{frame}[t]{Equilibrio beta: segni corretti}
\small

Per la cattura elettronica
\[
p+e^-\ \rightleftharpoons\ n+\nu_e,
\]
la condizione e
\[
\mu_p+\mu_e=\mu_n+\mu_{\nu_e}.
\]
Per il decadimento $n\rightleftharpoons p+e^-+\bar\nu_e$:
\[
\mu_n=\mu_p+\mu_e+\mu_{\bar\nu_e},
\qquad\mu_{\bar\nu_e}=-\mu_{\nu_e}.
\]
Per neutrini liberi che sfuggono da una stella fredda,
\[
\boxed{\mu_{\nu_e}=0\quad\Longrightarrow\quad\mu_n=\mu_p+\mu_e.}
\]
Per particella/antiparticella $\mu_{\bar i}=-\mu_i$ quando il potenziale e associato alle corrispondenti cariche; fotoni in equilibrio chimico ordinario: $\mu_\gamma=0$.
\end{frame}

\begin{frame}[t]{Potenziali chimici delle cariche conservate}
\small

Quando le reazioni rapide conservano $B,Q,L$:
\[
\boxed{\mu_i=B_i\mu_B+Q_i\mu_Q+L_i\mu_L.}
\]
Esempi:
\[
\begin{array}{c|ccc|c}
 i&B_i&Q_i&L_i&\mu_i\\\hline
 n&1&0&0&\mu_B\\
 p&1&1&0&\mu_B+\mu_Q\\
 e^-&0&-1&1&-\mu_Q+\mu_L\\
 \nu_e&0&0&1&\mu_L
\end{array}
\]
In stella fredda trasparente ai neutrini: $\mu_L=0$ (in questo modello); ne segue $\mu_e=-\mu_Q$, \emph{non} $\mu_e=0$.
In proto-stelle con neutrini intrappolati si deve invece tenere conto delle frazioni leptoniche conservate sui tempi dinamici.
\end{frame}

\begin{frame}[t]{Neutralita elettrica e composizione}
\small

Per $P=P(T,\mu_B,\mu_Q,\mu_L)$:
\[
n_Q=\left.\frac{\partial P}{\partial\mu_Q}\right|_{T,\mu_B,\mu_L}
=\sum_iQ_i n_i.
\]
Neutralita bulk:
\[
\boxed{n_Q=0.}
\]
Nella miscela $n,p,e^-$:
\[
n_Q=n_p-n_e=0.
\]
La soluzione per $\mu_Q$ dipende dall'EOS: in presenza di interazioni, $n_i$ e campi medi sono interdipendenti. Si impongono simultaneamente neutralita, equilibri chimici ed equazioni del modello microscopico.
\end{frame}

\begin{frame}[t]{Esempio di vincoli: npe in equilibrio}
\small

Assumiamo $T=0$, neutrini liberi e gas ideale per ciascuna specie:
\[
\mu_n=\mu_p+\mu_e,\qquad n_p=n_e,\qquad n_B=n_n+n_p.
\]
Per fermioni di spin $1/2$:
\[
n_i=\frac{k_{F,i}^3}{3\pi^2},\qquad
\mu_i=\sqrt{k_{F,i}^2+m_i^2}.
\]
La neutralita implica $k_{F,p}=k_{F,e}$, ma i potenziali chimici restano diversi perche $m_p\ne m_e$.
\[
\sqrt{k_{F,n}^2+m_n^2}=\sqrt{k_{F,p}^2+m_p^2}
 +\sqrt{k_{F,e}^2+m_e^2}.
\]
Il modello e didattico: non incorpora il legame nucleare ne le interazioni in materia densa.
\end{frame}

\section{3.3 Materia in beta-equilibrio}

\begin{frame}[t]{Soglia di neutronizzazione nel gas libero}
\small

Prima della comparsa dei neutroni, $k_{F,n}=0$ e $k_{F,e}=k_{F,p}\equiv k_F$:
\[
\sqrt{k_F^2+m_e^2}+\sqrt{k_F^2+m_p^2}=m_n.
\]
Dato che $k_F\ll m_p$ alla soglia,
\[
\sqrt{k_F^2+m_p^2}\simeq m_p,
\qquad
k_{F,c}\simeq\sqrt{(m_n-m_p)^2-m_e^2}.
\]
La densita critica (per elettroni o protoni) e
\[
\boxed{n_c\simeq\frac{[(m_n-m_p)^2-m_e^2]^{3/2}}{3\pi^2(\hbar c)^3}.}
\]
\emph{Correzione al sorgente:} sotto radice compare un segno meno, non piu. $m_n-m_p\simeq1.293\,\mathrm{MeV}$; $m_p\simeq938.272\,\mathrm{MeV}$.
\end{frame}

\begin{frame}[t]{Limite ad alta densita del gas libero npe}
\small

Nel limite formale $k_{F,i}\gg m_i$ per tutte le specie, non alla soglia,
\[
\mu_n=\mu_p+\mu_e\quad\Longrightarrow\quad
k_{F,n}\simeq k_{F,p}+k_{F,e}.
\]
Neutralita $k_{F,e}=k_{F,p}$:
\[
k_{F,n}\simeq2k_{F,p},\qquad
\frac{n_n}{n_p}\simeq\left(\frac{k_{F,n}}{k_{F,p}}\right)^3=8.
\]
\[
\boxed{\frac{n_n}{n_B}\simeq\frac89,\qquad\frac{n_p}{n_B}\simeq\frac19.}
\]
Questo \`e un limite matematico del gas libero, non una previsione affidabile della frazione protonica nel core reale: le interazioni nucleari e i muoni modificano la composizione.
\end{frame}

\begin{frame}[t]{Perche la materia reale forma nuclei?}
\small

Nel gas libero si trascura il guadagno di energia dovuto al legame nucleare.
\begin{itemize}
\item In crosta esterna: nuclei ricchi di neutroni in reticolo coulombiano + elettroni degeneri.
\item La composizione dipende dall'energia di massa nucleare, dall'energia elettronica e dall'energia del reticolo.
\item Aumentando la densita, il costo energetico del gas di elettroni favorisce catture elettroniche.
\item Al \emph{neutron drip} iniziano a comparire neutroni non legati nei nuclei.
\end{itemize}
La minimizzazione al numero barionico e alla neutralita fissati richiede confrontare varie specie $(A,Z)$, non solo scegliere il nucleo con massimo $E_b/A$.
\end{frame}

\begin{frame}[t]{Strati della crosta: scale orientative}
\small

\begin{itemize}
\item $\rho\lesssim10^4\,\mathrm{g\,cm^{-3}}$: atmosfere / materia atomica.
\item $\rho\sim10^4\,\mathrm{g\,cm^{-3}}$: nuclei e elettroni; la cristallizzazione dipende anche dalla temperatura.
\item $\rho\sim10^6\!-\!10^7\,\mathrm{g\,cm^{-3}}$: elettroni sempre piu relativistici; composizione nucleare cambia.
\item $\rho_{\rm drip}\sim4\times10^{11}\,\mathrm{g\,cm^{-3}}$: neutroni liberi e crosta interna.
\end{itemize}
Nel testo il ruolo delle chiusure di shell ($N=50,82$) e rilevante; le soglie numeriche dipendono dai modelli nucleari e dalle condizioni termiche.
\end{frame}

\section{3.4 Equazione di stato}

\begin{frame}[t]{Tre forme dell'equazione di stato}
\small

\[
P=P(\varepsilon),\qquad P=P(n_B),\qquad
\frac EA(n_B)=\frac{\varepsilon(n_B)}{n_B}.
\]
L'energia di legame per barione e
\[
\frac{E_b}{A}=\frac{\varepsilon}{n_B}-m_N
\]
(con la convenzione di riferimento per $m_N$ specificata).
\begin{itemize}
\item TOV richiede $\varepsilon(P)$; $P(\varepsilon)$ e sufficiente per la massa e il raggio.
\item Le grandezze barioniche e l'energia di legame richiedono anche $n_B$ e una normalizzazione dell'energia.
\item Per materia fredda in equilibrio: $P=n_B\mu_B-\varepsilon$, $\mu_B=d\varepsilon/dn_B$.
\end{itemize}
\end{frame}

\begin{frame}[t]{Derivazione: P dalla energia per barione}
\small

Poniamo $e(n_B)=\varepsilon(n_B)/n_B$; allora
\[
\frac{de}{dn_B}
=\frac1{n_B}\frac{d\varepsilon}{dn_B}-\frac{\varepsilon}{n_B^2}
=\frac{\mu_B n_B-\varepsilon}{n_B^2}.
\]
Da $P=\mu_B n_B-\varepsilon$:
\[
\boxed{P(n_B)=n_B^2\frac{de}{dn_B}.}
\]
Se $e=m_N+E_b/A$, la derivata di $m_N$ e nulla:
\[
P=n_B^2\frac{d}{dn_B}\left(\frac{E_b}{A}\right).
\]
A saturazione della materia nucleare simmetrica $de/dn_B=0\Rightarrow P=0$; non e una proprieta generica della materia neutronica in equilibrio beta.
\end{frame}

\section{3.5 Gas di Fermi liberi}

\begin{frame}[t]{Densita, energia e pressione dal gas degenere}
\small

Fermioni liberi isotropi, degenerazione $g$ e $T=0$:
\[
n=\frac{g}{(2\pi)^3}\int_{|\mathbf k|<k_F}d^3k=\frac{gk_F^3}{6\pi^2}.
\]
Con $E(k)=\sqrt{k^2+m^2}$,
\[
\varepsilon=\frac{g}{2\pi^2}\int_0^{k_F}k^2 E(k)\,dk,
\]
\[
\boxed{P=\frac{g}{6\pi^2}\int_0^{k_F}\frac{k^4}{E(k)}\,dk.}
\]
Interpretazione microscopica: $P=\frac13\langle\mathbf p\cdot\mathbf v\rangle$ per distribuzione isotropa, con $\mathbf v=\partial E/\partial\mathbf p=\mathbf p/E$.
\end{frame}

\begin{frame}[t]{Gas non relativistico: esponente 5/3}
\small

Per $k_F\ll m$: $E\simeq m$, $v\simeq k/m$. Quindi
\[
P_{\rm NR}=\frac{g}{6\pi^2m}\int_0^{k_F}k^4\,dk
=\frac{gk_F^5}{30\pi^2m}.
\]
Eliminando $k_F$ tramite $n=gk_F^3/(6\pi^2)$:
\[
\boxed{P_{\rm NR}=\frac1{5m}\left(\frac{6\pi^2}{g}\right)^{2/3}n^{5/3}.}
\]
L'energia di massa domina $\varepsilon\simeq mn$ (piu una correzione cinetica $\sim n^{5/3}$). La formula vale per elettroni, neutroni o protoni liberi nel rispettivo regime NR.
\end{frame}

\begin{frame}[t]{Nana bianca: elettroni NR e pressione}
\small

Neutralita per nuclei $(A,Z)$:
\[
n_e=Zn_A=\frac ZA n_B,\qquad \rho\simeq m_Nn_B.
\]
\[
P_e\simeq\frac1{5m_e}(3\pi^2)^{2/3}n_e^{5/3}
=\frac{(3\pi^2)^{2/3}}{5m_e}
\left(\frac{Z\rho}{Am_N}\right)^{5/3}.
\]
\[
\boxed{P\propto\rho^{5/3}.}
\]
La pressione e prevalentemente elettronica, mentre la massa/energia e prevalentemente nucleonica. Corretta la distinzione $\varepsilon\simeq m_A n_A\simeq m_N n_B$, non $m_A\rho_A$.
\end{frame}

\begin{frame}[t]{Elettroni ultrarelativistici: esponente 4/3}
\small

Se $k_{F,e}\gg m_e$ ma i nuclei rimangono NR,
\[
P_e\simeq\frac{g_e}{6\pi^2}\int_0^{k_F}k^3\,dk
=\frac{g_e k_F^4}{24\pi^2}.
\]
\[
\boxed{P_e=\frac14\left(\frac{6\pi^2}{g_e}\right)^{1/3}n_e^{4/3};\quad g_e=2.}
\]
Sostituendo $n_e=(Z/A)\rho/m_N$:
\[
P\propto\rho^{4/3}.
\]
Qui $\varepsilon$ totale della materia e ancora dominata dalla massa a riposo dei nuclei, se le approssimazioni di composizione e regime restano applicabili.
\end{frame}

\begin{frame}[t]{Gas di neutroni liberi: limite NR}
\small

Con $g_n=2$:
\[
n_n=\frac{k_{F,n}^3}{3\pi^2},\qquad
P_n^{\rm NR}=\frac{k_{F,n}^5}{15\pi^2m_n}.
\]
Quindi
\[
\boxed{P_n^{\rm NR}=\frac{(3\pi^2)^{2/3}}{5m_n}n_n^{5/3},\quad
\varepsilon\simeq m_n n_n.}
\]
L'esponente $5/3$ si applica a una singola specie fermionica non relativistica.
Per una stella di neutroni reale il potenziale nucleone--nucleone non e trascurabile e l'EOS puo essere molto diversa.
\end{frame}

\begin{frame}[t]{Gas di neutroni liberi: limite UR}
\small

Nel limite \emph{formale} $k_{F,n}\gg m_n$:
\[
P_n^{\rm UR}=\frac{g_n}{6\pi^2}\int_0^{k_F}k^3\,dk
=\frac{g_n k_F^4}{24\pi^2},
\]
\[
\varepsilon_n^{\rm UR}=\frac{g_n}{2\pi^2}\int_0^{k_F}k^3\,dk
=\frac{g_n k_F^4}{8\pi^2}.
\]
\[
\boxed{P_n^{\rm UR}=\frac13\varepsilon_n^{\rm UR}.}
\]
Dunque $P\propto n^{4/3}$ ma $P\propto\varepsilon^1$, perche $\varepsilon$ non e piu proporzionale a $m_n n$.
\end{frame}

\begin{frame}[t]{Gas relativistico completo: $P(k_F)$}
\small

Scriviamo $E_F=\sqrt{k_F^2+m^2}$ e $L=\ln[(k_F+E_F)/m]$.
\[
P=\frac{g}{6\pi^2}\int_0^{k_F}\frac{k^4}{\sqrt{k^2+m^2}}\,dk.
\]
L'integrale fornisce
\[
\boxed{P=\frac{g}{48\pi^2}\left[k_FE_F(2k_F^2-3m^2)+3m^4L\right].}
\]
Verifiche:
\[
P\xrightarrow{k_F\ll m}\frac{gk_F^5}{30\pi^2m},\qquad
P\xrightarrow{k_F\gg m}\frac{gk_F^4}{24\pi^2}.
\]
Nel sorgente originale era errato il coefficiente della forma integrale per $P$, non l'idea dell'integrazione.
\end{frame}

\begin{frame}[t]{Gas relativistico completo: energia e $n(k_F)$}
\small

\[
\varepsilon=\frac{g}{2\pi^2}\int_0^{k_F}k^2\sqrt{k^2+m^2}\,dk.
\]
Calcolando la primitiva:
\[
\boxed{\varepsilon=\frac{g}{16\pi^2}\left[k_FE_F(2k_F^2+m^2)-m^4L\right],}
\]
con $L=\ln[(k_F+E_F)/m]$.
La densita numerica rimane
\[
\boxed{n=\frac{gk_F^3}{6\pi^2}.}
\]
Per un gas ideale a $T=0$ si puo controllare
\[
\mu=E_F,\qquad \varepsilon+P=\mu n.
\]
L'EOS parametrica e la coppia $(\varepsilon(k_F),P(k_F))$.
\end{frame}

\begin{frame}[t]{Indici politropici: due definizioni da distinguere}
\small

\[
P\propto n^\Gamma,\qquad P\propto\varepsilon^\gamma.
\]
\[
\begin{array}{l|c|c}
\text{regime}&\Gamma&\gamma\\\hline
\text{gas fermionico NR}&5/3&5/3\ (\varepsilon\simeq mn)\\
\text{elettroni UR, nuclei NR}&4/3&4/3\ (\varepsilon\simeq m_N n_B)\\
\text{gas di una specie UR}&4/3&1\ (\varepsilon=3P)
\end{array}
\]
Il rapporto $P/\varepsilon$ e l'indice $d\ln P/d\ln n$ sono concetti distinti. Il secondo puo variare con la densita e con la composizione.
\end{frame}

\section{3.6 Politropiche}

\begin{frame}[t]{Politropica: costruzione termodinamica}
\small

Partiamo da $P=Kn^\Gamma$ e dall'identita
\[
P=n^2\frac{d}{dn}\left(\frac\varepsilon n\right).
\]
Se $\Gamma\ne1$:
\[
\frac{d}{dn}\left(\frac\varepsilon n\right)=Kn^{\Gamma-2}
\ \Longrightarrow\ \frac\varepsilon n=c+\frac{K}{\Gamma-1}n^{\Gamma-1}.
\]
\[
\boxed{\varepsilon(n)=cn+\frac{K}{\Gamma-1}n^\Gamma.}
\]
Per una politropica valida fino a $n\to0$ e $\Gamma>1$ si puo porre $c=m$; per segmenti raccordati a densita finita $c$ e fissata dal punto di raccordo, non automaticamente da $m$.
\end{frame}

\begin{frame}[t]{Potenziale chimico e velocita del suono}
\small

\[
\mu=\frac{d\varepsilon}{dn}=c+\frac{\Gamma K}{\Gamma-1}n^{\Gamma-1},
\qquad P=Kn^\Gamma.
\]
\[
\frac{dP}{dn}=\Gamma K n^{\Gamma-1}=\frac{\Gamma P}{n},
\qquad\frac{d\varepsilon}{dn}=\frac{\varepsilon+P}{n}.
\]
Percio
\[
\boxed{c_s^2=\frac{dP}{d\varepsilon}
=\frac{dP/dn}{d\varepsilon/dn}
=\frac{\Gamma P}{\varepsilon+P}.}
\]
Nelle unita $c=1$, la condizione locale di causalita e $0\le c_s^2\le1$. Nel sorgente originale era scritto il prodotto delle derivate anziche il loro rapporto.
\end{frame}

\begin{frame}[t]{Limite ad alta densita e cautela su Gamma=2}
\small

Per $P=Kn^\Gamma$, $\varepsilon=cn+P/(\Gamma-1)$:
\[
c_s^2=\frac{\Gamma P}{cn+\Gamma P/(\Gamma-1)}.
\]
Se $\Gamma>1$ e il termine $P$ domina su $cn$:
\[
\boxed{c_s^2\longrightarrow\Gamma-1.}
\]
Per $\Gamma>2$ la prosecuzione asintotica e acausale.
Per $\Gamma=2$ si ha $c_s^2\to1$ \emph{nel limite} di alta densita; non $P=\varepsilon$ esattamente a densita finita, perche $\varepsilon=cn+P$.
L'EOS di un gas di Fermi UR da invece $c_s^2=1/3$.
\end{frame}

\begin{frame}[t]{Raccordo di politropiche e transizioni}
\small

Per un raccordo liscio alla densita $n_*$ fra segmenti con diverso $\Gamma$:
\[
P_1(n_*)=P_2(n_*),\qquad
\mu_1(n_*)=\mu_2(n_*).
\]
Poiche $\varepsilon+P=n\mu$, segue la continuita di $\varepsilon$ nel raccordo a $n_*$.
\begin{itemize}
\item I parametri $K$ e le costanti $c$ sono determinati dal raccordo.
\item Un salto di densita a $P$ e $\mu$ fissi (transizione del primo ordine) va trattato separatamente; non deriva da un raccordo liscio.
\item Una curva $P(n)$ costruita a tratti richiede controlli di stabilita e causalita in ogni segmento.
\end{itemize}
\end{frame}

\section{Superfluidita neutronica: estensione oltre il capitolo 3}

\subsection{Fenomenologia e canali di pairing}

\begin{frame}[t]{Perche i neutroni possono essere superfluidi?}
\small

Neutroni: fermioni neutri di spin $1/2$. Un canale attrattivo efficace vicino alla superficie di Fermi puo generare \emph{pairing} tra stati correlati nel tempo inverso.
\[
\langle n_{-\mathbf k\downarrow}n_{\mathbf k\uparrow}\rangle\ne0
\quad\Longrightarrow\quad
\text{rottura spontanea di }U(1)_B\text{ (approssimata).}
\]
\begin{itemize}
\item Superfluidita: trasporto coerente del numero barionico e vortici quantizzati.
\item Non equivale a superconduttivita elettromagnetica: la coppia $nn$ e elettricamente neutra, quindi niente espulsione Meissner per il condensato neutronico.
\item Nelle stelle di neutroni si distinguono pairing neutronico e pairing protonico; il secondo e carico ed e un problema distinto.
\end{itemize}
Lo stato BCS descrive il condensato fermionico, non un gas diluito di molecole $nn$ reali.
\end{frame}

\begin{frame}[t]{Canali di pairing nel sistema nn}
\small

Per due neutroni identici, la funzione d'onda totale deve essere antisimmetrica allo scambio. Con isospin $T=1$ simmetrico, vale $(-1)^{L+S}=-1$.
\[
{}^{2S+1}L_J:
\quad {}^1S_0\ (L=0,S=0,J=0),\qquad {}^3P_2\ (L=1,S=1,J=2).
\]
\begin{itemize}
\item Crosta interna e zona di bassa densita: neutroni soprattutto nel canale isotropo ${}^1S_0$.
\item A densita maggiori nel core: canale tripletto ${}^3P_2$, spesso accoppiato a ${}^3F_2$; gap angolarmente anisotropo.
\item La localizzazione dei canali e le dimensioni dei gap dipendono da interazione, massa efficace, polarizzazione del mezzo e composizione.
\end{itemize}
Da qui in poi deriviamo l'equazione BCS prima per il canale singoletto isotropo.
\end{frame}

\begin{frame}[t]{Stato a due neutroni: simmetria spaziale e spin}
\small

Centro di massa $\mathbf K=\mathbf k_1+\mathbf k_2$, momento relativo $\mathbf k=(\mathbf k_1-\mathbf k_2)/2$:
\[
\mathbf k_1=\mathbf K/2+\mathbf k,\quad
\mathbf k_2=\mathbf K/2-\mathbf k.
\]
Per dispersione libera $\epsilon_k=\hbar^2k^2/(2m_n)$:
\[
\epsilon_{k_1}+\epsilon_{k_2}
=\frac{\hbar^2K^2}{4m_n}+\frac{\hbar^2k^2}{m_n}.
\]
Il pairing ordinario omogeneo favorisce $\mathbf K=0$.
Nel canale ${}^1S_0$ la parte spaziale e simmetrica e lo spin e singoletto:
\[
|S=0,M_S=0\rangle=
\frac{|\uparrow\downarrow\rangle-|\downarrow\uparrow\rangle}{\sqrt2}.
\]
L'antisimmetria riguarda la funzione d'onda \emph{totale}.
\end{frame}

\subsection{Dal problema di Cooper alla teoria BCS}

\begin{frame}[t]{Problema di Cooper: Hamiltoniana proiettata}
\small

Due neutroni sopra un mare di Fermi ideale, con $|\mathbf k|>k_F$:
\[
|\Psi\rangle=\sum_{\mathbf k>k_F}g_{\mathbf k}
|\mathbf k,-\mathbf k;{}^1S_0\rangle.
\]
Proiettando $(H_0+V_{nn})|\Psi\rangle=E|\Psi\rangle$:
\[
\boxed{(E-2\epsilon_k)g_{\mathbf k}=
\sum_{\mathbf k'>k_F}V_{\mathbf k\mathbf k'}g_{\mathbf k'},}
\]
\[
V_{\mathbf k\mathbf k'}=
\langle\mathbf k;{}^1S_0|V_{nn}|\mathbf k';{}^1S_0\rangle.
\]
$V_{\mathbf k\mathbf k'}$ e un elemento di matrice di scattering di coppie nel canale considerato, non un'interazione tra singoli neutroni indipendenti.
\end{frame}

\begin{frame}[t]{Modello didattico: attrazione costante in una shell}
\small

Solo per illustrare l'instabilita di Cooper scegliamo
\[
V_{\mathbf k\mathbf k'}=-V_0\quad(V_0>0)
\quad\text{entro }0<\xi_k,\xi_{k'}<\omega_c,
\]
con $\xi_k=\epsilon_k-E_F$.
La soluzione $g_k\propto(2\epsilon_k-E)^{-1}$ conduce a
\[
1=V_0\sum_{\mathbf k>k_F}\frac1{2\epsilon_k-E}.
\]
Ponendo $E=2E_F-E_b$ e densita di stati per spin $N(0)$ costante,
\[
\frac1{V_0N(0)}
\simeq\int_0^{\omega_c}\frac{d\xi}{2\xi+E_b}
=\frac12\ln\frac{2\omega_c+E_b}{E_b}.
\]
Per $E_b\ll\omega_c$: $E_b\simeq2\omega_c e^{-2/[V_0N(0)]}$.
\end{frame}

\begin{frame}[t]{Limiti dell'analogia con i superconduttori}
\small

Nel modello elettronico elementare l'interazione attrattiva puo essere attiva in una finestra di energie associata ai fononi del reticolo.
\begin{itemize}
\item Il \emph{problema di Cooper} dimostra che il mare di Fermi rende instabile lo stato normale rispetto al pairing per attrazione debole in una shell.
\item In materia neutronica l'interazione e $V_{nn}$, proiettata nel canale ${}^1S_0$ o ${}^3P_2$; non si introduce \emph{fisicamente} un cutoff Debye.
\item Per una EOS e gap realistici occorrono dispersione e interazione nel mezzo, possibilmente non locali e dipendenti da densita.
\end{itemize}
Il problema di una coppia aggiunta al mare di Fermi non e ancora la teoria BCS auto-consistente di molte coppie sovrapposte.
\end{frame}

\begin{frame}[t]{Seconda quantizzazione: operatori di coppia}
\small

Siano $a_{\mathbf k\sigma}^{\dagger}$ gli operatori che creano neutroni con impulso $\mathbf k$ e spin $\sigma$.
Una coppia in singoletto:
\[
b_{\mathbf k}^{\dagger}=
\frac{a_{\mathbf k\uparrow}^{\dagger}a_{-\mathbf k\downarrow}^{\dagger}
-a_{\mathbf k\downarrow}^{\dagger}a_{-\mathbf k\uparrow}^{\dagger}}{\sqrt2}.
\]
Hamiltoniana ridotta nel canale ${}^1S_0$:
\[
H_{\rm int}^{{}^1S_0}
=\sum_{\mathbf k\mathbf k'}
\langle\mathbf k;{}^1S_0|V_{nn}|\mathbf k';{}^1S_0\rangle
 b_{\mathbf k}^{\dagger}b_{\mathbf k'}.
\]
Con una convenzione che evita il doppio conteggio dei momenti $\pm\mathbf k$; fattori di normalizzazione possono essere assorbiti negli elementi di matrice.
\end{frame}

\begin{frame}[t]{Stato variazionale BCS}
\small

Su un insieme non duplicato di coppie di stati time-reversed:
\[
\boxed{|\mathrm{BCS}\rangle=
\prod_{\mathbf k}\left(u_k+v_k
 a_{\mathbf k\uparrow}^{\dagger}a_{-\mathbf k\downarrow}^{\dagger}\right)|0\rangle,}
\]
\[
|u_k|^2+|v_k|^2=1.
\]
Ogni coppia di stati contiene una sovrapposizione coerente tra occupazione vuota e doppia:
\[
\langle a_{\mathbf k\uparrow}^{\dagger}a_{\mathbf k\uparrow}\rangle=v_k^2,
\quad\langle a_{-\mathbf k\downarrow}a_{\mathbf k\uparrow}\rangle=u_kv_k
\]
(con fasi scelte reali).
Lo stato BCS non ha $N$ fissato esattamente: descrive l'equilibrio gran canonico; lo stato a $N$ fissato si puo ottenere proiettando il numero.
\end{frame}

\begin{frame}[t]{Decoupling di campo medio e definizione del gap}
\small

Scriviamo $\xi_k=\epsilon_k-\mu_n$, includendo nella $\epsilon_k$ un eventuale spettro di particella singola efficace.
Per il pairing omogeneo singoletto:
\[
\boxed{\Delta_{\mathbf k}=-\sum_{\mathbf k'}V_{\mathbf k\mathbf k'}
\langle a_{-\mathbf k'\downarrow}a_{\mathbf k'\uparrow}\rangle.}
\]
Il termine quartico di interazione viene linearizzato rispetto alla media anomala:
\[
H_{\rm MF}-\mu N=\sum_{\mathbf k}\xi_k
(a_{\mathbf k\uparrow}^{\dagger}a_{\mathbf k\uparrow}
+a_{-\mathbf k\downarrow}^{\dagger}a_{-\mathbf k\downarrow})
-\sum_{\mathbf k}(\Delta_k a_{\mathbf k\uparrow}^{\dagger}
 a_{-\mathbf k\downarrow}^{\dagger}+\mathrm{h.c.})+C.
\]
$C$ rende coerente l'energia di campo medio; la convenzione di segno di $\Delta$ e arbitraria globalmente.
\end{frame}

\begin{frame}[t]{Diagonalizzazione Bogoliubov: matrice 2x2}
\small

Nella base di Nambu $\Psi_k=(a_{\mathbf k\uparrow},a_{-\mathbf k\downarrow}^{\dagger})^T$ compare
\[
\mathcal H_k=\begin{pmatrix}\xi_k&-\Delta_k\\
-\Delta_k^*&-\xi_k\end{pmatrix}.
\]
Dall'equazione agli autovalori:
\[
\det(\mathcal H_k-EI)=
(\xi_k-E)(-\xi_k-E)-|\Delta_k|^2=0.
\]
\[
\boxed{E_k=\sqrt{\xi_k^2+|\Delta_k|^2},\qquad\lambda_\pm=\pm E_k.}
\]
Le eccitazioni elementari sono quasiparticelle di Bogoliubov, miscele coerenti di neutroni e lacune; $E_k$ ha minimo circa $|\Delta(k_F)|$ per un gap isotropo non nullo.
\end{frame}

\begin{frame}[t]{Coefficienti di coerenza u e v}
\small

Dagli autovettori di $\mathcal H_k$ e dalla normalizzazione $|u_k|^2+|v_k|^2=1$ si trova
\[
\boxed{u_k^2=\frac12\left(1+\frac{\xi_k}{E_k}\right),\quad
v_k^2=\frac12\left(1-\frac{\xi_k}{E_k}\right).}
\]
Per $\Delta_k$ reale positivo:
\[
u_kv_k=\frac{\Delta_k}{2E_k}.
\]
Lontano dalla superficie di Fermi $v_k^2$ tende alla funzione gradino normale; in prossimita di $k_F$ l'occupazione viene smussata su una scala energetica controllata da $\Delta$.
A $T>0$ compare $1-2f(E_k)=\tanh(E_k/(2T))$.
\end{frame}

\begin{frame}[t]{Equazione di gap BCS: materia neutronica}
\small

Usando la media anomala a temperatura $T$:
\[
\langle a_{-\mathbf k\downarrow}a_{\mathbf k\uparrow}\rangle
=\frac{\Delta_k}{2E_k}\tanh\left(\frac{E_k}{2T}\right).
\]
In limite termodinamico:
\[
\boxed{\Delta(\mathbf k)=-\int\frac{d^3k'}{(2\pi)^3}
V_{{}^1S_0}(\mathbf k,\mathbf k')
\frac{\Delta(\mathbf k')}{2E(\mathbf k')}
\tanh\frac{E(\mathbf k')}{2T}.}
\]
\[
E(k)=\sqrt{[\epsilon(k)-\mu_n]^2+|\Delta(k)|^2}.
\]
A $T=0$, $\tanh(E/2T)\to1$. La legge dipende dall'interazione di coppia neutronica e non contiene un cutoff fononico postulato.
\end{frame}

\begin{frame}[t]{Proiezione sferica del canale 1S0}
\small

Se il sistema e isotropo e $\Delta(\mathbf k)=\Delta(k)$, si puo proiettare l'interazione su onde parziali:
\[
V_0(k,k')=\frac12\int_{-1}^{1}d\cos\theta\,
V(\mathbf k,\mathbf k')
\]
(per una convenzione di normalizzazione dell'elemento di matrice).
Poiche $d^3k'=4\pi k'^2dk'$ dopo media angolare,
\[
\boxed{\Delta(k)=-\frac1{2\pi^2}\int_0^\infty
k'^2dk'\,V_0(k,k')\frac{\Delta(k')}{2E(k')}
\tanh\frac{E(k')}{2T}.}
\]
Il prefattore dipende dalla convenzione con cui si definisce $V_0$; l'equazione integrale tridimensionale precedente ne fissa il significato fisico.
\end{frame}

\begin{frame}[t]{Equazione del numero e autoconsistenza}
\small

La densita dei neutroni va calcolata con la distribuzione BCS, non identificando $\mu_n$ a priori con $E_F$ di gas libero:
\[
\boxed{n_n=\int\frac{d^3k}{(2\pi)^3}
\left[1-\frac{\xi_k}{E_k}\tanh\left(\frac{E_k}{2T}\right)\right].}
\]
A $T=0$ questa formula coincide con $n_n=2\int d^3k\,v_k^2/(2\pi)^3$.
Il problema fisico si risolve imponendo simultaneamente:
\begin{itemize}
\item equazione di gap per $\Delta(k)$;
\item equazione del numero per $\mu_n$;
\item se necessario, spettro $\epsilon(k)$ e massa efficace calcolati autoconsistentemente.
\end{itemize}
La EOS della materia stellare deve inoltre rispettare neutralita e $\beta$-equilibrio.
\end{frame}

\begin{frame}[t]{Temperatura critica: modello debole isotropo}
\small

Nel modello BCS con attrazione costante e densita di stati per spin $N(0)$:
\[
\Delta(0)\simeq2\omega_c e^{-1/[N(0)V_0]},\quad
T_c\simeq\frac{e^{\gamma_E}}\pi\Delta(0).
\]
Quindi
\[
\boxed{\frac{2\Delta(0)}{T_c}\simeq3.53\quad
(\text{BCS isotropo, debole accoppiamento}).}
\]
Qui $\gamma_E\simeq0.577$ e la costante di Eulero--Mascheroni.
La relazione non e universale nei canali anisotropi ${}^3P_2$ e puo cambiare per forti correlazioni del mezzo; $\omega_c$ e solo un parametro del \emph{modello}, non una frequenza Debye della materia neutronica.
\end{frame}

\begin{frame}[t]{Energia di condensazione e scala spaziale}
\small

Per accoppiamento debole, gas uniforme, $T=0$, gap isotropo:
\[
\frac{E_{\rm SF}-E_{\rm normal}}{V}
\simeq-\frac12 N(0)\Delta(0)^2
\]
con $N(0)$ densita di stati \emph{per spin} alla superficie di Fermi.
La scala di coerenza della coppia e
\[
\boxed{\xi_0\sim\frac{\hbar v_F}{\pi\Delta(k_F)}.}
\]
\begin{itemize}
\item Gap piccolo rispetto a $E_F$: coppie estese e sovrapposte, non molecole puntiformi.
\item $\Delta$ e il gap \emph{di una quasiparticella}; creare due quasiparticelle costa almeno circa $2\Delta$ nel modello isotropo ideale.
\item Nella crosta il reticolo nucleare puo modificare mobilita e densita superfluida (entrainment).
\end{itemize}
\end{frame}


% --- Microfisica della crosta: collegamenti complementari ---
\begin{frame}[t]{Livelli nucleonici, shell e correzioni alla descrizione semiclassica}
\small
\begin{itemize}
\item La descrizione semiclassica sostituisce la somma su livelli discreti con un integrale sulla densità degli stati; la differenza oscillante è la correzione di shell.
\item In un campo medio $h\varphi_\alpha=\epsilon_\alpha\varphi_\alpha$, la posizione dei livelli dipende da potenziale, spin--orbita e geometria del cluster.
\item I minimi energetici presso chiusure di shell possono cambiare il numero protonico preferito nelle celle della crosta interna.
\item La microfisica fornisce quindi un contributo alle differenze di energia fra configurazioni che un modello a goccia liquida liscio non risolve.
\end{itemize}
{\footnotesize Collegamento a Haensel et al., §3.4.3; la trattazione della EOS resta in NS.tex.}
\end{frame}

\begin{frame}[t]{Fononi del reticolo nucleare: modi collettivi}
\small
Per piccoli spostamenti $\mathbf u_i$ dei nuclei dalle posizioni di equilibrio, l'energia armonica è
\[
U=U_0+\frac12\sum_{ij,ab}u_i^a D_{ij}^{ab}u_j^b.
\]
\begin{itemize}
\item La diagonalizzazione della matrice dinamica fornisce le frequenze dei fononi $\omega_s(\mathbf q)$.
\item I modi trasversali, nel limite a lunga lunghezza d'onda, sono controllati dal modulo di taglio; il settore longitudinale risente anche di compressibilità e screening.
\item I fononi contribuiscono a calore specifico e processi di scattering, ma nella crosta neutronica possono accoppiarsi ai modi del superfluido.
\end{itemize}
{\footnotesize Raccordo microscopico con Haensel et al., §3.7.}
\end{frame}

\begin{frame}[t]{Reticolo, neutroni superfluidi ed entrainment}
\small
\begin{itemize}
\item I neutroni non legati della crosta interna si muovono in un potenziale periodico generato dai cluster; la loro mobilità dipende dalla struttura a bande.
\item L'\emph{entrainment} indica che corrente di particelle e velocità delle diverse componenti non sono in generale proporzionali in modo indipendente.
\item In una formulazione lineare a due componenti si può scrivere schematicamente
\[
\mathbf j_n=\rho_{nn}\mathbf v_n+\rho_{np}\mathbf v_p,
\qquad
\mathbf j_p=\rho_{pn}\mathbf v_n+\rho_{pp}\mathbf v_p.
\]
\item Le matrici di risposta devono rispettare simmetrie e vincoli di conservazione; i coefficienti dipendono da densità, pairing e reticolo.
\item Conseguenze: inerzia dinamica della crosta, modi collettivi e bilancio di momento angolare nei glitch.
\end{itemize}
{\footnotesize Approfondimento di raccordo: struttura quantistica della crosta e superfluidità.}
\end{frame}
% --- fine integrazione microphysics ---

\begin{frame}[t]{Neutroni 3P2 e complessita del core}
\small

Per accoppiamento tripletto in ${}^3P_2$ (con miscela ${}^3F_2$), il gap dipende in generale anche dalla direzione $\hat{\mathbf k}$:
\[
\Delta\to\Delta_{ab}(\mathbf k),\qquad
E(\mathbf k)=\sqrt{\xi_k^2+\text{autovalori della matrice di gap}}.
\]
\begin{itemize}
\item Lo stato d'ordine non e riducibile in generale a un singolo scalare $\Delta(k)$.
\item Alcune fasi possono avere minima angolari molto diverse o nodi; le leggi di soppressione termica non sono sempre esponenziali.
\item Le previsioni dipendono da interazione tensoriale, polarizzazione del mezzo, forze a tre corpi e composizione.
\end{itemize}
Pertanto i frame ${}^1S_0$ precedenti costituiscono la derivazione BCS di riferimento, non una soluzione completa del core profondo.
\end{frame}

\subsection{Fase, vortici e osservabili}

\begin{frame}[t]{Fase del condensato e velocita superfluida}
\small

L'ordine di coppia e complesso:
\[
\Delta(\mathbf r)=|\Delta(\mathbf r)|e^{i\theta(\mathbf r)}.
\]
Per un condensato neutro di coppie con massa $m_{\rm pair}\simeq2m_n$, il flusso superfluido nel limite non relativistico e
\[
\boxed{\mathbf v_s=\frac{\hbar}{2m_n}\nabla\theta.}
\]
Poiche $e^{i\theta}$ deve essere monodromo lungo una curva chiusa:
\[
\oint\nabla\theta\cdot d\boldsymbol\ell=2\pi\ell,
\qquad\ell\in\mathbb Z.
\]
La circolazione e quantizzata:
\[
\boxed{\oint\mathbf v_s\cdot d\boldsymbol\ell=
\ell\frac h{2m_n}.}
\]
Questa e la firma idrodinamica di superfluidita neutronica; non coinvolge un campo gauge elettromagnetico.
\end{frame}

\begin{frame}[t]{Vortici e rotazione delle stelle di neutroni}
\small

Una superfluidita irrotazionale in regioni prive di vortici:
\[
\nabla\times\mathbf v_s=0\quad
(\text{dove la fase }\theta\text{ e regolare}).
\]
La rotazione macroscopica si realizza con una distribuzione di vortici quantizzati. Per vortici rettilinei e rotazione quasi uniforme:
\[
\boxed{n_v\simeq\frac{2\Omega}{\kappa},\qquad
\kappa=\frac{h}{2m_n},}
\]
con $n_v$ densita areale e $\Omega$ velocita angolare del superfluido.
\begin{itemize}
\item Pinning dei vortici al reticolo della crosta e loro dinamica possono contribuire ai glitch.
\item La mobilita effettiva e l'entrainment dipendono dalla struttura nucleare della crosta.
\item In RS/RG completa bisogna generalizzare la descrizione idrodinamica e distinguere quadrivelocita del normale e corrente superfluida.
\end{itemize}
\end{frame}

\begin{frame}[t]{Calore specifico, raffreddamento e pair breaking}
\small

Uno spettro isotropo con gap $\Delta>0$ riduce le eccitazioni termiche a $T\ll T_c$:
\[
n_{\rm qp}(T)\propto e^{-\Delta/T}\times
\text{fattore di potenza in }T.
\]
\begin{itemize}
\item Capacita termica dei quasineutroni: ridotta sotto $T_c$.
\item Processi neutrino che richiedono quasiparticelle nucleoniche: spesso soppressi dal pairing a bassa $T$.
\item Vicino all'inizio del pairing, \emph{pair breaking and formation} (PBF) puo invece contribuire all'emissione di neutrini.
\item Nel caso anisotropo con nodi il comportamento puo diventare una legge di potenza, non puramente esponenziale.
\end{itemize}
Per un confronto osservativo serve il profilo $\Delta(n,T)$ e un modello termico stellare; non basta il solo BCS omogeneo.
\end{frame}

\subsection{Landau--Ginzburg neutra: utilita e limiti}

\begin{frame}[t]{Perche una breve Landau--Ginzburg per neutroni?}
\small

La teoria microscopica BCS calcola pairing e quasiparticelle. Vicino a $T_c$ e su scale spaziali lunghe si puo invece usare un campo efficace complesso $\Psi(\mathbf r)$ associato all'ordine di coppia.
\[
\boxed{F[\Psi]-F_N=\int d^3r\left[
 a(T)|\Psi|^2+\frac b2|\Psi|^4+K|\nabla\Psi|^2\right].}
\]
\begin{itemize}
\item $b>0$, $K>0$, $a(T)=a_0(T-T_c)$ per una transizione continua.
\item In un superfluido \emph{neutro} compare $\nabla$, non il derivato elettromagnetico $\nabla-iq\mathbf A$ della superconduttivita.
\item La normalizzazione di $\Psi$ e convenzionale: $|\Psi|^2$ non va identificato automaticamente con la densita di neutroni superfluidi.
\end{itemize}
Applicazione: profili dell'ordine e struttura dei vortici vicino a $T_c$, non EOS neutronica del core freddo.
\end{frame}

\begin{frame}[t]{Minimizzazione GL omogenea e lunghezza di coerenza}
\small

Per campo uniforme $\nabla\Psi=0$:
\[
f-f_N=a|\Psi|^2+\frac b2|\Psi|^4.
\]
Ponendo $x=|\Psi|^2$ e derivando,
\[
\frac{df}{dx}=a+bx=0
\ \Longrightarrow\ x=-\frac ab\quad(a<0).
\]
\[
\boxed{|\Psi_0|^2=-\frac{a(T)}b\propto(T_c-T),\quad T<T_c.}
\]
La minimizzazione funzionale fornisce
\[
\boxed{-K\nabla^2\Psi+a\Psi+b|\Psi|^2\Psi=0.}
\]
Una scala caratteristica del recupero spaziale e $\xi_{\rm GL}\sim\sqrt{K/|a(T)|}$, che diverge quando $T\to T_c^-$.
\end{frame}

\begin{frame}[t]{Validita relativa di BCS e GL per stelle di neutroni}
\small

\begin{itemize}
\item BCS/HFB: necessario per stimare gap $\Delta(k,n)$, $T_c(n)$ e quasiparticelle a partire da $V_{nn}$; utile per crosta e core neutronici.
\item GL neutra: espansione \emph{vicino a una transizione continua} e su lunghezze $\gg$ quelle microscopiche; puo chiarire profili e difetti topologici.
\item Il core freddo di una stella matura puo essere lontano da $T_c$; GL scalare non sostituisce BCS ne descrive in generale il canale anisotropo ${}^3P_2$.
\item Per fenomenologia quantitativa di glitch/raffreddamento servono EoS, entrainment, struttura dei vortici e trasporto oltre il funzionale minimo.
\end{itemize}
Per questi motivi l'approfondimento GL e intenzionalmente breve e interamente formulato per \emph{superfluidita neutra}.
